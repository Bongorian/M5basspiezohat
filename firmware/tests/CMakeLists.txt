cmake_minimum_required(VERSION 3.16)
project(bass_core_tests LANGUAGES CXX)
enable_testing()
add_executable(bass_core_tests test_core.cpp ../components/bass_core/bass_core.cpp ../components/bass_core/midi_descriptor.cpp)
target_include_directories(bass_core_tests PRIVATE ../components/bass_core/include)
target_compile_features(bass_core_tests PRIVATE cxx_std_17)
target_compile_options(bass_core_tests PRIVATE -Wall -Wextra -Werror -O2)
option(BASS_SANITIZERS "Address and undefined behavior sanitizers" ON)
if(BASS_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(bass_core_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bass_core_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME bass_core COMMAND bass_core_tests)
add_executable(bass_optimization_tests test_optimization.cpp ../components/bass_core/bass_core.cpp)
target_include_directories(bass_optimization_tests PRIVATE ../components/bass_core/include)
target_compile_features(bass_optimization_tests PRIVATE cxx_std_17)
target_compile_options(bass_optimization_tests PRIVATE -Wall -Wextra -Werror -O2)
if(BASS_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(bass_optimization_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bass_optimization_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME bass_optimization COMMAND bass_optimization_tests)
add_executable(bass_core_benchmark benchmark_core.cpp ../components/bass_core/bass_core.cpp)
target_include_directories(bass_core_benchmark PRIVATE ../components/bass_core/include)
target_compile_features(bass_core_benchmark PRIVATE cxx_std_17)
target_compile_options(bass_core_benchmark PRIVATE -Wall -Wextra -Werror -O2)
target_compile_definitions(bass_core_benchmark PRIVATE BASS_BENCH_PROFILES=1)
# Timing uses the optimized unsanitized binary; correctness is tested above.
